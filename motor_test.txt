// ESP32 + TB6612FNG - Two Motor Test

// Motor A connections
#define AIN1 14
#define AIN2 27
#define PWMA 26

// Motor B connections
#define BIN1 18
#define BIN2 19
#define PWMB 25

// Standby pin
#define STBY 13

// PWM settings
#define PWM_FREQ 1000
#define PWM_RESOLUTION 8

void setup() {
  Serial.begin(115200);

  // Configure motor pins
  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);

  pinMode(BIN1, OUTPUT);
  pinMode(BIN2, OUTPUT);

  pinMode(STBY, OUTPUT);

  // Enable motor driver
  digitalWrite(STBY, HIGH);



  // Initially stop both motors
  stopMotors();

  Serial.println("Two Motor Test Started");
}

// Motor A control
void motorA(int speed) {
  speed = constrain(speed, -255, 255);

  if (speed > 0) {
    digitalWrite(AIN1, HIGH);
    digitalWrite(AIN2, LOW);
  }
  else if (speed < 0) {
    digitalWrite(AIN1, LOW);
    digitalWrite(AIN2, HIGH);
  }
  else {
    digitalWrite(AIN1, LOW);
    digitalWrite(AIN2, LOW);
  }

  ledcWrite(PWMA, abs(speed));
}

// Motor B control
void motorB(int speed) {
  speed = constrain(speed, -255, 255);

  if (speed > 0) {
    digitalWrite(BIN1, HIGH);
    digitalWrite(BIN2, LOW);
  }
  else if (speed < 0) {
    digitalWrite(BIN1, LOW);
    digitalWrite(BIN2, HIGH);
  }
  else {
    digitalWrite(BIN1, LOW);
    digitalWrite(BIN2, LOW);
  }

  ledcWrite(PWMB, abs(speed));
}

// Move both motors forward
void forward(int speed) {
  motorA(speed);
  motorB(speed);
}

// Move both motors backward
void backward(int speed) {
  motorA(-speed);
  motorB(-speed);
}

// Stop both motors
void stopMotors() {
  motorA(0);
  motorB(0);
}

void loop() {

  // Forward
  Serial.println("BOTH MOTORS FORWARD");
  forward(180);
  delay(2000);

  // Stop
  Serial.println("BOTH MOTORS STOP");
  stopMotors();
  delay(1000);

  // Backward
  Serial.println("BOTH MOTORS BACKWARD");
  backward(180);
  delay(2000);

  // Stop
  Serial.println("BOTH MOTORS STOP");
  stopMotors();
  delay(1000);
}
