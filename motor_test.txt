// ESP32 + TB6612FNG - Single Motor Test

// Motor A connections
#define AIN1 14
#define AIN2 27
#define PWMA 26

// Standby pin
#define STBY 13

// PWM settings
#define PWM_FREQ 1000
#define PWM_RESOLUTION 8

void setup() {
  Serial.begin(115200);

  // Configure motor pins
  pinMode(AIN1, OUTPUT);
  pinMode(AIN2, OUTPUT);
  pinMode(STBY, OUTPUT);

  // Enable motor driver
  digitalWrite(STBY, HIGH);

  // Configure PWM for ESP32
  

  // Initially stop motor
  stopMotor();

  Serial.println("Single Motor Test Started");
}

// Motor control function
void motorA(int speed) {
  speed = constrain(speed, -255, 255);

  if (speed > 0) {
    // Forward
    digitalWrite(AIN1, HIGH);
    digitalWrite(AIN2, LOW);
  }
  else if (speed < 0) {
    // Backward
    digitalWrite(AIN1, LOW);
    digitalWrite(AIN2, HIGH);
  }
  else {
    // Stop
    digitalWrite(AIN1, LOW);
    digitalWrite(AIN2, LOW);
  }

  ledcWrite(PWMA, abs(speed));
}

// Stop motor
void stopMotor() {
  motorA(0);
}

void loop() {

  // Forward
  Serial.println("Motor FORWARD");
  motorA(180);
  delay(2000);

  // Stop
  Serial.println("Motor STOP");
  stopMotor();
  delay(1000);

  // Backward
  Serial.println("Motor BACKWARD");
  motorA(-180);
  delay(2000);

  // Stop
  Serial.println("Motor STOP");
  stopMotor();
  delay(1000);
}
